  <script>
    let ttsActive = true;
    let knownScores = {}; // Tracks game state: { gameId: { homeScore: X, awayScore: Y } }
    let isInitialLoad = true;

    function toggleTTS() {
      ttsActive = !ttsActive;
      const btn = document.getElementById('ttsBtn');
      btn.innerText = ttsActive ? '🔊 Voice Alerts ON' : '🔇 Voice Alerts OFF';
      btn.className = ttsActive 
        ? 'text-xs font-semibold px-3 py-1.5 rounded-full bg-emerald-950/50 border border-emerald-500/50 text-emerald-300'
        : 'text-xs font-semibold px-3 py-1.5 rounded-full bg-slate-800 border border-slate-700 text-slate-400';
    }

    // -------------------------------------------------------------
    // LIVE SCORE AUTO-POLLING (ESPN API)
    // -------------------------------------------------------------
    async function fetchLiveScores() {
      const league = document.getElementById('leagueSelect').value;
      const url = `https://site.api.espn.com/apis/site/v2/sports/${league}/scoreboard`;

      try {
        const response = await fetch(url);
        const data = await response.json();
        renderGames(data.events || []);
      } catch (err) {
        console.warn('Live score fetch failed:', err);
        document.getElementById('gamesContainer').innerHTML = 
          `<div class="text-center py-6 text-sm text-red-400">Failed to load live sports feed. Retrying shortly...</div>`;
      }
    }

    function renderGames(events) {
      const container = document.getElementById('gamesContainer');
      if (!events || events.length === 0) {
        container.innerHTML = `<div class="text-center py-6 text-sm text-slate-500">No scheduled games found for this league today.</div>`;
        return;
      }

      container.innerHTML = '';

      events.forEach(event => {
        const gameId = event.id;
        const competition = event.competitions[0];
        const home = competition.competitors.find(c => c.homeAway === 'home');
        const away = competition.competitors.find(c => c.awayAway === 'away');

        const homeScore = parseInt(home.score || '0');
        const awayScore = parseInt(away.score || '0');
        const homeColor = '#' + (home.team.color || '00B2FF');
        const awayColor = '#' + (away.team.color || 'FFC845');

        // Check for score increases after initial baseline load
        if (!isInitialLoad && knownScores[gameId]) {
          if (homeScore > knownScores[gameId].home) {
            triggerScoreEvent(home.team.displayName, homeColor);
          } else if (awayScore > knownScores[gameId].away) {
            triggerScoreEvent(away.team.displayName, awayColor);
          }
        }

        // Store current scores
        knownScores[gameId] = { home: homeScore, away: awayScore };

        const statusText = event.status.type.shortDetail || 'Scheduled';
        const isLive = event.status.type.state === 'in';

        // Render game item card
        const card = document.createElement('div');
        card.className = "p-4 rounded-lg bg-slate-950/70 border border-slate-800 flex items-center justify-between";
        card.innerHTML = `
          <div class="flex items-center space-x-3 w-2/5">
            <span class="h-4 w-4 rounded-full flex-shrink-0" style="background-color: ${awayColor}"></span>
            <span class="font-semibold text-slate-200 truncate">${away.team.shortDisplayName}</span>
            <span class="text-lg font-bold text-slate-100 ml-auto">${awayScore}</span>
          </div>
          <div class="text-center px-2">
            <span class="text-[10px] font-mono uppercase px-2 py-0.5 rounded bg-slate-800 text-slate-400">${event.status.type.name}</span>
            <div class="text-xs ${isLive ? 'text-cyan-400 font-semibold' : 'text-slate-500'} mt-1">${statusText}</div>
          </div>
          <div class="flex items-center space-x-3 w-2/5 justify-end">
            <span class="text-lg font-bold text-slate-100 mr-auto">${homeScore}</span>
            <span class="font-semibold text-slate-200 truncate">${home.team.shortDisplayName}</span>
            <span class="h-4 w-4 rounded-full flex-shrink-0" style="background-color: ${homeColor}"></span>
          </div>
        `;
        container.appendChild(card);
      });

      isInitialLoad = false;
    }

    // -------------------------------------------------------------
    // SMART LIGHT API TRIGGERS
    // -------------------------------------------------------------
    async function sendGoveeTrigger(key, mac, model) {
      if (!key || !mac || !model) return;
      try {
        await fetch('https://developer-api.govee.com/v1/devices/control', {
          method: 'PUT',
          headers: { 'Govee-API-Key': key, 'Content-Type': 'application/json' },
          body: JSON.stringify({ device: mac, model: model, cmd: { name: "color", value: { r: 0, g: 178, b: 255 } } })
        });
      } catch (e) { console.warn('Govee API failed:', e); }
    }

    async function sendNanoleafTrigger(ip, token) {
      if (!ip || !token) return;
      try {
        await fetch(`http://${ip}:16021/api/v1/${token}/state`, {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ hue: { value: 180 }, sat: { value: 100 }, brightness: { value: 100 } })
        });
      } catch (e) { console.warn('Nanoleaf API failed:', e); }
    }

    async function sendLifxTrigger(token, selector) {
      if (!token) return;
      try {
        await fetch(`https://api.lifx.com/v1/lights/${selector || 'all'}/effects/pulse`, {
          method: 'POST',
          headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' },
          body: JSON.stringify({ color: 'cyan', cycles: 5, period: 0.5, power_on: true })
        });
      } catch (e) { console.warn('LIFX API failed:', e); }
    }

    async function sendWledTrigger(ip) {
      if (!ip) return;
      try {
        await fetch(`http://${ip}/json/state`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ on: true, bri: 255, seg: [{ col: [[0, 178, 255]], fx: 9 }] })
        });
      } catch (e) { console.warn('WLED API failed:', e); }
    }

    async function sendHueTrigger(ip, user) {
      if (!ip || !user) return;
      try {
        await fetch(`http://${ip}/api/${user}/lights/1/state`, {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ on: true, bri: 254, hue: 35000, sat: 254, alert: 'lselect' })
        });
      } catch (e) { console.warn('Hue API failed:', e); }
    }

    function triggerScoreEvent(teamName, teamColor) {
      // Voice alert
      if (ttsActive && 'speechSynthesis' in window) {
        const msg = new SpeechSynthesisUtterance(`Score alert! ${teamName} just scored!`);
        window.speechSynthesis.speak(msg);
      }

      // Flash background UI with team color
      document.body.style.backgroundColor = teamColor || '#004C54';
      setTimeout(() => { document.body.style.backgroundColor = '#0b0f19'; }, 1500);

      // Flash Govee
      if (document.getElementById('goveeActive').checked) {
        sendGoveeTrigger(
          document.getElementById('goveeApiKey').value,
          document.getElementById('goveeMac').value,
          document.getElementById('goveeModel').value
        );
      }

      // Flash Nanoleaf
      if (document.getElementById('nanoleafActive').checked) {
        sendNanoleafTrigger(
          document.getElementById('nanoleafIp').value,
          document.getElementById('nanoleafAuthToken').value
        );
      }

      // Flash LIFX
      if (document.getElementById('lifxActive').checked) {
        sendLifxTrigger(
          document.getElementById('lifxToken').value,
          document.getElementById('lifxSelector').value
        );
      }

      // Flash WLED
      if (document.getElementById('wledActive').checked) {
        sendWledTrigger(document.getElementById('wledIp').value);
      }

      // Flash Philips Hue
      if (document.getElementById('hueActive').checked) {
        sendHueTrigger(
          document.getElementById('hueIp').value,
          document.getElementById('hueUser').value
        );
      }
    }

    // Initialize initial poll and repeat every 10 seconds
    fetchLiveScores();
    setInterval(fetchLiveScores, 10000);
  </script>
